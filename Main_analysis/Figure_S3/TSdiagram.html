depth = df['depth']
station = df['Station']
station_unique = df2['station']
lat = df_latlon['Latitude']
lon = df_latlon['Longitude']
sp = df['Salinity']
pt = df['PotTemperature']

# Derive/convert quantities
index2 = np.zeros(np.size(df.index))
lat2 = np.zeros(np.size(df.index))
lon2 = np.zeros(np.size(df.index))
for i in df2.index:
    station_profile = station_unique[i]
    index = df[df['Station'] == station_profile].index
    index2[index] = i
    lon2[index] = lon[i]
    lat2[index] = lat[i]
df["Station2"] = index2
df["lon"] = lon2
df["lat"] = lat2
p = gsw.p_from_z(depth,lat2)
sa = gsw.SA_from_SP(sp, p, lon2, lat2)
ct = gsw.CT_from_pt(sa,pt)

# Figure out boudaries (mins and maxs)
smin = sa.min() #- (0.01 * sa.min())
smax = sa.max() #+ (0.01 * sa.max())
tmin = ct.min() #- (0.1 * pt.max())
tmax = ct.max() #+ (0.1 * pt.max())

# Calculate how many gridcells we need in the x and y dimensions
xdim = round((smax-smin)/0.1+1,0)
ydim = round((tmax-tmin)+1,0)
    
xdim = int(xdim)
ydim = int(ydim)

# Create empty grid of zeros
dens = np.zeros((ydim,xdim))

# Create temp and salt vectors of appropiate dimensions
ti = np.linspace(0,ydim,ydim)+tmin
si = np.linspace(0,xdim,xdim)*0.1+smin

# derive freezing temperature
CT_freezing = gsw.CT_freezing([smin, smax], [np.nanmin(p),np.nanmin(p)],0)
                                  
# Loop to fill in grid with densities
for j in range(0,int(ydim)):
    for i in range(0, int(xdim)):
        dens[j,i]=gsw.rho(si[i],ti[j],0)

# Substract 1000 to convert to sigma-t
dens = dens - 1000


fig1 = plt.figure()
ax1 = fig1.add_subplot(111)
CS = plt.contour(si,ti,dens, linestyles='dashed', colors='k')
plt.clabel(CS, fontsize=12, inline=1, fmt='%1.1f') # Label every second level  
ax1.plot([smin,smax],CT_freezing,'k')

cax = ax1.scatter(sa, ct, s = 30, c = df["Station2"], cmap = plt.get_cmap('Spectral', 20), vmin = 0, vmax =20)

cmap =sns.color_palette("Spectral", n_colors=20)
labels =[]
for i in df2.index:
    station_profile = station_unique[i]
    index = df[df['Station'] == station_profile].index
    sa_profile = sa[index]
    ct_profile = ct[index]
    cax1 = ax1.plot(sa_profile, ct_profile, color=cmap[i])
    labels.append(str(station_unique.values[i]))
    
cb = plt.colorbar(cax)

cb.set_ticks(ticks= np.arange(20)+0.5, labels=labels)
cb.set_label('Station')
ax1.set_xlabel('Absolute Salinity [g kg$^{-1}$]')
ax1.set_ylabel('Conservative Temperature [$^\circ$C]')

ax1.text(-0.12, 1.05, 'a', transform=ax1.transAxes,
            size=20, weight='bold')

ax1.set_xlim([27,37])
ax1.set_ylim([-2,22])

plt.savefig('FIG_TSA.png', dpi = 300, bbox_inches='tight')
plt.savefig('FIG_TSA.pdf', dpi = 300, bbox_inches='tight')

plt.show()
plt.close()
